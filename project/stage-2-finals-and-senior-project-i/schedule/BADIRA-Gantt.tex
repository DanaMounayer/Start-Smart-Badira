  \ganttgroup[group/.append style={fill=#1,draw=#1},
              group left peak/.append style={fill=#1,draw=#1},
              group right peak/.append style={fill=#1,draw=#1},
              group label font=\bfseries\scriptsize\color{cInk}]{#2}{#3}{#4}}
